# Как создать приложение "Библиотека" на C# Windows Forms

## ШАГ 1: Создание проекта в Visual Studio

1. Открыть Visual Studio
2. Файл -> Создать -> Проект
3. Выбрать "Приложение Windows Forms (.NET)" или "Windows Forms App"
4. Назвать проект `CollegeLibrary`
5. Нажать "Создать"

---

## ШАГ 2: Создать класс Book.cs (модель данных)

Правой кнопкой по проекту -> Добавить -> Класс -> Назвать `Book.cs`

```csharp
using System;

namespace CollegeLibrary
{
    public class Book
    {
        public int Id { get; set; }
        public string Title { get; set; }
        public string Author { get; set; }
        public int Year { get; set; }
        public bool IsAvailable { get; set; }

        public Book()
        {
            Title = "";
            Author = "";
            IsAvailable = true;
        }
    }
}
```

**Что это:** Класс для хранения данных об одной книге.

## ШАГ 3: Создать класс Database.cs (работа с файлом)

Добавить -> Класс -> Назвать `Database.cs`

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;

namespace CollegeLibrary
{
    public class Database
    {
        private string filePath = "books.json";
        public List<Book> Books { get; set; }

        public Database()
        {
            Books = new List<Book>();
            Load();
        }

        public void Load()
        {
            if (File.Exists(filePath))
            {
                string json = File.ReadAllText(filePath);
                Books = JsonSerializer.Deserialize<List<Book>>(json);
                if (Books == null)
                    Books = new List<Book>();
            }
        }

        public void Save()
        {
            var options = new JsonSerializerOptions { WriteIndented = true };
            string json = JsonSerializer.Serialize(Books, options);
            File.WriteAllText(filePath, json);
        }

        public void AddBook(Book book)
        {
            int maxId = 0;
            foreach (var b in Books)
            {
                if (b.Id > maxId) maxId = b.Id;
            }
            book.Id = maxId + 1;
            Books.Add(book);
            Save();
        }

        public void DeleteBook(int id)
        {
            Book bookToRemove = null;
            foreach (var b in Books)
            {
                if (b.Id == id)
                {
                    bookToRemove = b;
                    break;
                }
            }
            if (bookToRemove != null)
            {
                Books.Remove(bookToRemove);
                Save();
            }
        }

        public List<Book> Search(string query)
        {
            List<Book> result = new List<Book>();
            query = query.ToLower();
            foreach (var b in Books)
            {
                if (b.Title.ToLower().Contains(query) ||
                    b.Author.ToLower().Contains(query))
                {
                    result.Add(b);
                }
            }
            return result;
        }
    }
}
```

**Что это:** Класс для сохранения/загрузки книг в JSON файл.

## ШАГ 4: Код главной формы (Form1.cs)

Открыть Form1.cs (или MainForm.cs) и заменить весь код:

```csharp
using System;
using System.Drawing;
using System.Windows.Forms;

namespace CollegeLibrary
{
    public partial class Form1 : Form
    {
        private DataGridView dataGrid;
        private TextBox txtTitle;
        private TextBox txtAuthor;
        private TextBox txtYear;
        private TextBox txtSearch;
        private Button btnAdd;
        private Button btnDelete;
        private Button btnSearch;
        private Button btnShowAll;
        private CheckBox chkAvailable;
        private Database db;

        public Form1()
        {
            db = new Database();
            InitializeMyComponents();
            LoadBooks();
        }

        private void InitializeMyComponents()
        {
            this.Text = "Библиотека колледжа";
            this.Size = new Size(800, 500);

            // Название
            Label lblTitle = new Label();
            lblTitle.Text = "Название:";
            lblTitle.Location = new Point(10, 15);
            this.Controls.Add(lblTitle);

            txtTitle = new TextBox();
            txtTitle.Location = new Point(85, 12);
            txtTitle.Size = new Size(150, 20);
            this.Controls.Add(txtTitle);

            // Автор
            Label lblAuthor = new Label();
            lblAuthor.Text = "Автор:";
            lblAuthor.Location = new Point(245, 15);
            this.Controls.Add(lblAuthor);

            txtAuthor = new TextBox();
            txtAuthor.Location = new Point(300, 12);
            txtAuthor.Size = new Size(120, 20);
            this.Controls.Add(txtAuthor);

            // Год
            Label lblYear = new Label();
            lblYear.Text = "Год:";
            lblYear.Location = new Point(430, 15);
            this.Controls.Add(lblYear);

            txtYear = new TextBox();
            txtYear.Location = new Point(470, 12);
            txtYear.Size = new Size(60, 20);
            this.Controls.Add(txtYear);

            // Доступна
            chkAvailable = new CheckBox();
            chkAvailable.Text = "Доступна";
            chkAvailable.Location = new Point(540, 12);
            chkAvailable.Checked = true;
            this.Controls.Add(chkAvailable);

            // Кнопка Добавить
            btnAdd = new Button();
            btnAdd.Text = "Добавить";
            btnAdd.Location = new Point(630, 10);
            btnAdd.Click += BtnAdd_Click;
            this.Controls.Add(btnAdd);

            // Поиск
            Label lblSearch = new Label();
            lblSearch.Text = "Поиск:";
            lblSearch.Location = new Point(10, 50);
            this.Controls.Add(lblSearch);

            txtSearch = new TextBox();
            txtSearch.Location = new Point(65, 47);
            txtSearch.Size = new Size(200, 20);
            this.Controls.Add(txtSearch);

            btnSearch = new Button();
            btnSearch.Text = "Найти";
            btnSearch.Location = new Point(275, 45);
            btnSearch.Click += BtnSearch_Click;
            this.Controls.Add(btnSearch);

            btnShowAll = new Button();
            btnShowAll.Text = "Показать все";
            btnShowAll.Location = new Point(355, 45);
            btnShowAll.Size = new Size(100, 25);
            btnShowAll.Click += BtnShowAll_Click;
            this.Controls.Add(btnShowAll);

            btnDelete = new Button();
            btnDelete.Text = "Удалить";
            btnDelete.Location = new Point(630, 45);
            btnDelete.Click += BtnDelete_Click;
            this.Controls.Add(btnDelete);

            // Таблица
            dataGrid = new DataGridView();
            dataGrid.Location = new Point(10, 80);
            dataGrid.Size = new Size(760, 370);
            dataGrid.AllowUserToAddRows = false;
            dataGrid.ReadOnly = true;
            dataGrid.SelectionMode = DataGridViewSelectionMode.FullRowSelect;
            this.Controls.Add(dataGrid);
        }

        private void LoadBooks()
        {
            dataGrid.DataSource = null;
            dataGrid.DataSource = db.Books;
        }

        private void BtnAdd_Click(object sender, EventArgs e)
        {
            if (string.IsNullOrEmpty(txtTitle.Text) ||
                string.IsNullOrEmpty(txtAuthor.Text))
            {
                MessageBox.Show("Заполните название и автора!");
                return;
            }

            int year = 0;
            if (!int.TryParse(txtYear.Text, out year))
            {
                MessageBox.Show("Введите корректный год!");
                return;
            }

            Book book = new Book();
            book.Title = txtTitle.Text;
            book.Author = txtAuthor.Text;
            book.Year = year;
            book.IsAvailable = chkAvailable.Checked;

            db.AddBook(book);

            txtTitle.Text = "";
            txtAuthor.Text = "";
            txtYear.Text = "";

            LoadBooks();
            MessageBox.Show("Книга добавлена!");
        }

        private void BtnDelete_Click(object sender, EventArgs e)
        {
            if (dataGrid.SelectedRows.Count == 0)
            {
                MessageBox.Show("Выберите книгу!");
                return;
            }

            int id = (int)dataGrid.SelectedRows[0].Cells["Id"].Value;

            if (MessageBox.Show("Удалить?", "Подтверждение",
                MessageBoxButtons.YesNo) == DialogResult.Yes)
            {
                db.DeleteBook(id);
                LoadBooks();
            }
        }

        private void BtnSearch_Click(object sender, EventArgs e)
        {
            if (string.IsNullOrEmpty(txtSearch.Text))
            {
                MessageBox.Show("Введите текст для поиска!");
                return;
            }

            var found = db.Search(txtSearch.Text);
            dataGrid.DataSource = null;
            dataGrid.DataSource = found;
        }

        private void BtnShowAll_Click(object sender, EventArgs e)
        {
            txtSearch.Text = "";
            LoadBooks();
        }
    }
}
```

## ЗАПОМНИ ГЛАВНОЕ:

1. **Модель** (Book.cs) - класс с данными (свойства)
2. **База данных** (Database.cs) - Load(), Save(), Add(), Delete()
3. **Форма** - создаёшь элементы в коде, добавляешь через `this.Controls.Add()`
4. **JSON** - `JsonSerializer.Serialize()` и `JsonSerializer.Deserialize()`
