# БИБЛИОТЕКА С АВТОРИЗАЦИЕЙ - ПОШАГОВАЯ ИНСТРУКЦИЯ

## ЧТО ДЕЛАЕТ ПРИЛОЖЕНИЕ:

1. Сначала окно ВХОДА (логин + пароль)
2. Можно зарегистрироваться или войти
3. После входа - окно с КНИГАМИ (добавить/удалить)

```

│   ВХОД В СИСТЕМУ    │         │  БИБЛИОТЕКА - Иван         │
│                     │         │                            │
│  Логин: [_____]     │  ──►    │  [Название книги] [Добав]  │
│  Пароль: [*****]    │         │  ┌──────────────┐  [Удал]  │
```

## ШАГ 4: LoginForm.cs

```csharp
using System;
using System.Collections.Generic;
using System.Drawing;
using System.IO;
using System.Windows.Forms;

namespace CollegeLibrary
{
    public class LoginForm : Form
    {
        private TextBox txtLogin;
        private TextBox txtPassword;
        private Button btnLogin;
        private Button btnRegister;

        private Dictionary<string, string> users = new Dictionary<string, string>();
        private string usersFile = "users.txt";

        public LoginForm()
        {
            CreateElements();
            LoadUsers();
        }

        private void CreateElements()
        {
            this.Text = "Вход в библиотеку";
            this.Size = new Size(300, 200);

            // Логин
            Label lblLogin = new Label();
            lblLogin.Text = "Логин:";
            lblLogin.Location = new Point(20, 20);
            this.Controls.Add(lblLogin);

            txtLogin = new TextBox();
            txtLogin.Location = new Point(100, 20);
            txtLogin.Size = new Size(150, 25);
            this.Controls.Add(txtLogin);

            // Пароль
            Label lblPass = new Label();
            lblPass.Text = "Пароль:";
            lblPass.Location = new Point(20, 55);
            this.Controls.Add(lblPass);

            txtPassword = new TextBox();
            txtPassword.Location = new Point(100, 55);
            txtPassword.Size = new Size(150, 25);
            txtPassword.PasswordChar = '*';
            this.Controls.Add(txtPassword);

            // Кнопки
            btnLogin = new Button();
            btnLogin.Text = "Войти";
            btnLogin.Location = new Point(50, 100);
            btnLogin.Click += BtnLogin_Click;
            this.Controls.Add(btnLogin);

            btnRegister = new Button();
            btnRegister.Text = "Регистрация";
            btnRegister.Location = new Point(150, 100);
            btnRegister.Size = new Size(100, 30);
            btnRegister.Click += BtnRegister_Click;
            this.Controls.Add(btnRegister);
        }

        private void LoadUsers()
        {
            if (File.Exists(usersFile))
            {
                string[] lines = File.ReadAllLines(usersFile);
                foreach (string line in lines)
                {
                    string[] parts = line.Split(';');
                    if (parts.Length == 2)
                    {
                        users[parts[0]] = parts[1];
                    }
                }
            }
        }

        private void SaveUsers()
        {
            List<string> lines = new List<string>();
            foreach (var user in users)
            {
                lines.Add(user.Key + ";" + user.Value);
            }
            File.WriteAllLines(usersFile, lines.ToArray());
        }

        private void BtnLogin_Click(object sender, EventArgs e)
        {
            string login = txtLogin.Text;
            string password = txtPassword.Text;

            if (users.ContainsKey(login) && users[login] == password)
            {
                this.Hide();
                BooksForm books = new BooksForm(login);
                books.ShowDialog();
                this.Close();
            }
            else
            {
                MessageBox.Show("Неверный логин или пароль!");
            }
        }

        private void BtnRegister_Click(object sender, EventArgs e)
        {
            string login = txtLogin.Text;
            string password = txtPassword.Text;

            if (users.ContainsKey(login))
            {
                MessageBox.Show("Пользователь уже есть!");
                return;
            }

            users[login] = password;
            SaveUsers();
            MessageBox.Show("Регистрация успешна!");
        }
    }
}
```

## ШАГ 5: BooksForm.cs

```csharp
using System;
using System.Collections.Generic;
using System.Drawing;
using System.IO;
using System.Windows.Forms;

namespace CollegeLibrary
{
    public class BooksForm : Form
    {
        private string currentUser;
        private List<string> books = new List<string>();
        private string fileName = "books.txt";

        private TextBox txtBook;
        private Button btnAdd;
        private Button btnDelete;
        private ListBox listBooks;

        public BooksForm(string username)
        {
            currentUser = username;
            CreateElements();
            LoadData();
        }

        private void CreateElements()
        {
            this.Text = "Библиотека - " + currentUser;
            this.Size = new Size(400, 350);

            txtBook = new TextBox();
            txtBook.Location = new Point(10, 10);
            txtBook.Size = new Size(250, 25);
            this.Controls.Add(txtBook);

            btnAdd = new Button();
            btnAdd.Text = "Добавить";
            btnAdd.Location = new Point(270, 10);
            btnAdd.Click += BtnAdd_Click;
            this.Controls.Add(btnAdd);

            btnDelete = new Button();
            btnDelete.Text = "Удалить";
            btnDelete.Location = new Point(270, 45);
            btnDelete.Click += BtnDelete_Click;
            this.Controls.Add(btnDelete);

            listBooks = new ListBox();
            listBooks.Location = new Point(10, 45);
            listBooks.Size = new Size(250, 250);
            this.Controls.Add(listBooks);
        }

        private void LoadData()
        {
            if (File.Exists(fileName))
            {
                string[] lines = File.ReadAllLines(fileName);
                books = new List<string>(lines);
            }
            UpdateList();
        }

        private void SaveData()
        {
            File.WriteAllLines(fileName, books.ToArray());
        }

        private void UpdateList()
        {
            listBooks.Items.Clear();
            foreach (string book in books)
            {
                listBooks.Items.Add(book);
            }
        }

        private void BtnAdd_Click(object sender, EventArgs e)
        {
            if (txtBook.Text != "")
            {
                books.Add(txtBook.Text);
                SaveData();
                UpdateList();
                txtBook.Text = "";
            }
        }

        private void BtnDelete_Click(object sender, EventArgs e)
        {
            if (listBooks.SelectedIndex >= 0)
            {
                books.RemoveAt(listBooks.SelectedIndex);
                SaveData();
                UpdateList();
            }
        }
    }
}
```
